function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var dataCSV = JSON.parse(e.postData.contents); if (!dataCSV || dataCSV.length === 0) { return ContentService.createTextOutput(JSON.stringify({"status": "gagal", "pesan": "Data kosong"})).setMimeType(ContentService.MimeType.JSON); } var barisTerakhir = Math.max(sheet.getLastRow(), 1); sheet.getRange(barisTerakhir + 1, 1, dataCSV.length, dataCSV[0].length).setValues(dataCSV); return ContentService.createTextOutput(JSON.stringify({"status": "sukses", "pesan": dataCSV.length + " baris data berhasil disimpan!"})).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"status": "gagal", "pesan": "Terjadi kesalahan: " + error.message})).setMimeType(ContentService.MimeType.JSON); } } function doOptions(e) { return ContentService.createTextOutput("").setMimeType(ContentService.MimeType.TEXT); }
